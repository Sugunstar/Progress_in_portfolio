import { addPropertyControls, ControlType, useIsStaticRenderer } from "framer"
import { motion, useSpring, useTransform, useMotionValue, useInView } from "framer-motion"
import {
    useEffect,
    useRef,
    useState,
    useCallback,
    useMemo,
    startTransition,
    type CSSProperties,
} from "react"

/* ---------- TYPES ---------- */
type Stage = 0 | 1 | 2

interface CursorMaskRevealProps {
    foregroundImage?: { src: string }
    backgroundImage?: { src: string }
    maskSize: number
    maskShape: "circle" | "square" | "rounded" | "custom"
    maskImage?: { src: string }
    followSpeed: number
    borderRadius: number
    initialX: number
    initialY: number
    mainHeadingLines: string[]
    mainDescription: string
    menuItems: string[]
    stage2MenuBackgroundImage?: { src: string }
    stage2MenuTextColor: string
    stage2MainTextColor: string
    stage2HeadingFont: any
    stage2BodyFont: any
    stage2ButtonFont: any
    introTitle: string
    cursorHintStage0Text: string
    cursorHintStage2Text: string
    cursorHintTextColor: string
    cursorHintBackgroundColor: string
    cursorHintBorderColor: string
    cursorHintFont: any
}

export default function CursorMaskReveal(props: CursorMaskRevealProps) {
    const {
        foregroundImage,
        backgroundImage,
        maskSize,
        maskShape,
        maskImage,
        followSpeed,
        borderRadius,
        initialX,
        initialY,
        mainHeadingLines,
        mainDescription,
        menuItems,
        stage2MenuBackgroundImage,
        stage2MenuTextColor,
        stage2MainTextColor,
        stage2HeadingFont,
        stage2BodyFont,
        stage2ButtonFont,
        introTitle,
        cursorHintStage0Text,
        cursorHintStage2Text,
        cursorHintTextColor,
        cursorHintBackgroundColor,
        cursorHintBorderColor,
        cursorHintFont,
    } = props

    const containerRef = useRef<HTMLDivElement>(null)
    const isStatic = useIsStaticRenderer()

    /* ---------- SIZE ---------- */
    const [containerSize, setContainerSize] = useState({ width: 0, height: 0 })
    const touchLastYRef = useRef<number | null>(null)
    const [hasScrolled, setHasScrolled] = useState(false)
    const [hasOpenedMenu, setHasOpenedMenu] = useState(false)
    const [isPointerInside, setIsPointerInside] = useState(false)
    const [showMask, setShowMask] = useState(false)
    const hideMaskTimeoutRef = useRef<number | null>(null)

    useEffect(() => {
        if (!containerRef.current) return
        const update = () => {
            const r = containerRef.current!.getBoundingClientRect()
            startTransition(() =>
                setContainerSize({ width: r.width, height: r.height })
            )
        }
        update()
        const ro = new ResizeObserver(update)
        ro.observe(containerRef.current)
        return () => ro.disconnect()
    }, [])

    /* ---------- SCROLL OFFSET ---------- */
    const offsetMV = useMotionValue(0)
    const [offsetState, setOffsetState] = useState(0)

    useEffect(() => {
        offsetMV.set(offsetState)
    }, [offsetState, offsetMV])

    /* ---------- STAGE ---------- */
    const stage: Stage =
        containerSize.height === 0
            ? 0
            : offsetState < containerSize.height * 0.6
              ? 0
              : offsetState < containerSize.height * 1.6
                ? 1
                : 2

    /* ---------- WHEEL ---------- */
    useEffect(() => {
        if (!containerRef.current || isStatic) return

        const onWheel = (e: WheelEvent) => {
            e.preventDefault()
            if (!containerSize.height) return

            const max = containerSize.height * 2
            setOffsetState((v) => Math.max(0, Math.min(max, v + e.deltaY)))
            startTransition(() => {
                setHasScrolled(true)
            })
        }

        const el = containerRef.current
        el.addEventListener("wheel", onWheel, { passive: false })
        return () => el.removeEventListener("wheel", onWheel)
    }, [containerSize.height, isStatic])

    /* ---------- CURSOR ---------- */
    const mouseX = useMotionValue(initialX)
    const mouseY = useMotionValue(initialY)

    const smoothX = useSpring(mouseX, {
        stiffness: 80 + followSpeed * 300,
        damping: 18,
    })
    const smoothY = useSpring(mouseY, {
        stiffness: 80 + followSpeed * 300,
        damping: 18,
    })

    const fullscreenSize =
        Math.max(containerSize.width, containerSize.height) * 2.5

    /* ✅ SAFE TRANSFORM (MotionValue → MotionValue) */
    const baseMaskSizeMV = useSpring(
        useTransform(
            offsetMV,
            [0, containerSize.height],
            [maskSize, fullscreenSize],
            { clamp: true }
        ),
        { stiffness: 120, damping: 22 }
    )

    // Spring that represents pointer presence (1 when inside, 0 when outside)
    const pointerPresence = useSpring(0, {
        stiffness: 200,
        damping: 30,
    })

    useEffect(() => {
        pointerPresence.set(isPointerInside ? 1 : 0)
    }, [isPointerInside, pointerPresence])

    const maskSizeMV = useTransform(
        [baseMaskSizeMV, pointerPresence],
        ([base, presence]) => base * presence,
    )

    const maskX = useTransform([smoothX, maskSizeMV], ([x, s]) => x - s / 2)
    const maskY = useTransform([smoothY, maskSizeMV], ([y, s]) => y - s / 2)

    const inverseX = useTransform(maskX, (v) => -v)
    const inverseY = useTransform(maskY, (v) => -v)

    const updatePointerFromClient = (clientX: number, clientY: number) => {
        if (!containerRef.current) return
        const r = containerRef.current.getBoundingClientRect()
        mouseX.set(clientX - r.left)
        mouseY.set(clientY - r.top)
    }

    const handleMouseMove = (e: React.MouseEvent) => {
        updatePointerFromClient(e.clientX, e.clientY)
    }

    const handleMouseEnter = () => {
        if (hideMaskTimeoutRef.current !== null && typeof window !== "undefined") {
            window.clearTimeout(hideMaskTimeoutRef.current)
            hideMaskTimeoutRef.current = null
        }
        startTransition(() => {
            setIsPointerInside(true)
            setShowMask(true)
        })
    }

    const handleMouseLeave = () => {
        startTransition(() => {
            setIsPointerInside(false)
        })
        if (typeof window !== "undefined") {
            const timeoutId = window.setTimeout(() => {
                startTransition(() => {
                    setShowMask(false)
                })
                hideMaskTimeoutRef.current = null
            }, 400)
            hideMaskTimeoutRef.current = timeoutId
        }
    }

    const handleTouchStart = (e: React.TouchEvent<HTMLDivElement>) => {
        if (!e.touches[0]) return
        touchLastYRef.current = e.touches[0].clientY
        updatePointerFromClient(e.touches[0].clientX, e.touches[0].clientY)
    }

    const handleTouchMove = (e: React.TouchEvent<HTMLDivElement>) => {
        if (!containerSize.height || !e.touches[0]) return
        const currentY = e.touches[0].clientY
        const lastY = touchLastYRef.current ?? currentY
        const deltaY = lastY - currentY
        touchLastYRef.current = currentY

        const max = containerSize.height * 2
        setOffsetState((v) => Math.max(0, Math.min(max, v + deltaY)))
        startTransition(() => {
            setHasScrolled(true)
        })
        updatePointerFromClient(e.touches[0].clientX, e.touches[0].clientY)
    }

    /* ---------- PAGE MOVE ---------- */
    const contentY = useSpring(
        useTransform(offsetMV, (v: number) =>
            v <= containerSize.height ? 0 : -(v - containerSize.height)
        ),
        { stiffness: 120, damping: 22 }
    )

    const getMaskStyle = (): CSSProperties => {
        if (maskShape === "circle") return { borderRadius: "50%" }
        if (maskShape === "square") return { borderRadius: 0 }
        if (maskShape === "rounded") return { borderRadius }
        if (maskShape === "custom" && maskImage?.src) {
            return {
                maskImage: `url(${maskImage.src})`,
                WebkitMaskImage: `url(${maskImage.src})`,
                maskRepeat: "no-repeat",
                WebkitMaskRepeat: "no-repeat",
                maskSize: "contain",
                WebkitMaskSize: "contain",
                maskPosition: "center",
                WebkitMaskPosition: "center",
            }
        }
        return { borderRadius: "50%" }
    }

    const fgSrc = foregroundImage?.src
    const bgSrc = backgroundImage?.src

    useEffect(() => {
        return () => {
            if (hideMaskTimeoutRef.current !== null && typeof window !== "undefined") {
                window.clearTimeout(hideMaskTimeoutRef.current)
            }
        }
    }, [])

    return (
        <div
            ref={containerRef}
            onMouseMove={isStatic ? undefined : handleMouseMove}
            onTouchStart={isStatic ? undefined : handleTouchStart}
            onTouchMove={isStatic ? undefined : handleTouchMove}
            onMouseEnter={isStatic ? undefined : handleMouseEnter}
            onMouseLeave={isStatic ? undefined : handleMouseLeave}
            style={{
                width: "100%",
                height: "100%",
                position: "relative",
                overflow: "hidden",
                background: "#000",
                cursor: "default",
            }}
        >
            <motion.div
                style={{
                    position: "absolute",
                    inset: 0,
                    height: "200%",
                    y: contentY,
                }}
            >
                {/* PAGE 1: Foreground + Background + Mask */}
                <div style={{ height: "50%", position: "relative" }}>
                    {bgSrc && (
                        <div
                            style={{
                                position: "absolute",
                                inset: 0,
                                backgroundImage: `url(${bgSrc})`,
                                backgroundSize: "cover",
                                backgroundPosition: "center",
                            }}
                        />
                    )}

                    {fgSrc && (
                        <motion.div
                            animate={{ opacity: stage === 0 ? 1 : 0 }}
                            transition={{ duration: 0.35 }}
                            style={{
                                position: "absolute",
                                inset: 0,
                                backgroundImage: `url(${fgSrc})`,
                                backgroundSize: "cover",
                                backgroundPosition: "center",
                                zIndex: 2,
                            }}
                        />
                    )}

                    {stage <= 1 && showMask && (
                        <motion.div
                            style={{
                                position: "absolute",
                                width: maskSizeMV,
                                height: maskSizeMV,
                                x: maskX,
                                y: maskY,
                                zIndex: 4,
                                overflow: "hidden",
                                pointerEvents: "none",
                                ...getMaskStyle(),
                            }}
                        >
                            <motion.div
                                style={{
                                    width: containerSize.width,
                                    height: containerSize.height,
                                    backgroundImage: `url(${bgSrc})`,
                                    backgroundSize: "cover",
                                    backgroundPosition: "center",
                                    x: inverseX,
                                    y: inverseY,
                                }}
                            />
                        </motion.div>
                    )}
                </div>

                {/* PAGE 2: Crazy Navigation */}
                <div style={{ height: "50%", position: "relative" }}>
                    <CrazyNavigation
                        menuLabel="Menu"
                        closeLabel="Close"
                        introTitle={introTitle}
                        menuItems={menuItems}
                        mainBackground="#000000"
                        menuBackground="#F5F0DD"
                        primaryTextColor={stage2MenuTextColor}
                        inverseTextColor={stage2MainTextColor}
                        headingFont={stage2HeadingFont}
                        bodyFont={stage2BodyFont}
                        buttonFont={stage2ButtonFont}
                        animationDuration={0.7}
                        mainHeadingLines={mainHeadingLines}
                        mainDescription={mainDescription}
                        stage={stage}
                        style={{ width: "100%", height: "100%" }}
                        stage2MenuBackgroundImage={stage2MenuBackgroundImage}
                        onMenuOpen={() =>
                            startTransition(() => {
                                setHasOpenedMenu(true)
                            })
                        }
                    />
                </div>
            </motion.div>
            {/* Cursor-following hints */}
            {!isStatic && isPointerInside && stage === 0 && !hasScrolled && (
                <motion.div
                    style={{
                        position: "absolute",
                        x: smoothX,
                        y: smoothY,
                        translateX: "-50%",
                        translateY: "-160%",
                        pointerEvents: "none",
                        zIndex: 999,
                        color: cursorHintTextColor,
                        textTransform: "uppercase",
                        background: cursorHintBackgroundColor,
                        padding: "8px 14px",
                        borderRadius: 999,
                        border: `1px solid ${cursorHintBorderColor}`,
                        whiteSpace: "nowrap",
                        boxShadow: "0 10px 30px rgba(0,0,0,0.35)",
                        backdropFilter: "blur(10px)",
                        ...cursorHintFont,
                    }}
                >
                    {cursorHintStage0Text}
                </motion.div>
            )}
            {!isStatic && isPointerInside && stage === 2 && !hasOpenedMenu && (
                <motion.div
                    style={{
                        position: "absolute",
                        x: smoothX,
                        y: smoothY,
                        translateX: "-50%",
                        translateY: "-160%",
                        pointerEvents: "none",
                        zIndex: 999,
                        color: cursorHintTextColor,
                        textTransform: "uppercase",
                        background: cursorHintBackgroundColor,
                        padding: "8px 14px",
                        borderRadius: 999,
                        border: `1px solid ${cursorHintBorderColor}`,
                        whiteSpace: "nowrap",
                        boxShadow: "0 10px 30px rgba(0,0,0,0.35)",
                        backdropFilter: "blur(10px)",
                        ...cursorHintFont,
                    }}
                >
                    {cursorHintStage2Text}
                </motion.div>
            )}
        </div>
    )
}

interface CrazyNavProps {
    menuLabel: string
    closeLabel: string
    introTitle: string
    menuItems: string[]
    mainBackground: string
    menuBackground: string
    primaryTextColor: string
    inverseTextColor: string
    headingFont: any
    bodyFont: any
    buttonFont: any
    animationDuration: number
    style?: CSSProperties
    children?: any
    mainHeadingLines: string[]
    mainDescription: string
    stage: Stage
    stage2MenuBackgroundImage?: { src: string }
    onMenuOpen?: () => void
}

// Starfield types
interface StarsSettings {
    count: number
    speed: number
    thickness: number
    opacity: number
    length: number
    thinning: number
    shortening: number
}

interface TravelInSpaceProps {
    backgroundColor: string
    radiusIn: number
    radiusOut: number
    stars: StarsSettings
    colors: string[]
    style?: CSSProperties
    clipRect?: { x: number; y: number; width: number; height: number }
}

interface Star {
    angle: number
    radius: number
    speedScale: number
}

function CrazyNavigation(props: CrazyNavProps) {
    const {
        menuLabel,
        closeLabel,
        introTitle,
        menuItems,
        mainBackground,
        menuBackground,
        primaryTextColor,
        inverseTextColor,
        headingFont,
        bodyFont,
        buttonFont,
        animationDuration,
        style,
        children,
        mainHeadingLines,
        mainDescription,
        stage,
        stage2MenuBackgroundImage,
        onMenuOpen,
    } = props

    const [isOpen, setIsOpen] = useState(false)
    const isStatic = useIsStaticRenderer()
    const panelRef = useRef<HTMLDivElement | null>(null)
    const headingRef = useRef<HTMLDivElement | null>(null)
    const [headingRect, setHeadingRect] = useState<{
        x: number
        y: number
        width: number
        height: number
    } | null>(null)
    const inView = useInView(panelRef, { margin: "0px", amount: 0.2 })

    const duration = useMemo(() => {
        if (isStatic || !inView) return 0
        return animationDuration
    }, [isStatic, inView, animationDuration])

    useEffect(() => {
        if (!panelRef.current || !headingRef.current) return

        const updateRect = () => {
            const panelRect = panelRef.current?.getBoundingClientRect()
            const headingBounds = headingRef.current?.getBoundingClientRect()
            if (!panelRect || !headingBounds) return

            const x = headingBounds.left - panelRect.left
            const y = headingBounds.top - panelRect.top
            const width = headingBounds.width
            const height = headingBounds.height

            startTransition(() => {
                // Narrow the clip area horizontally but keep full height
                const horizontalInsetFactor = 0.7
                const verticalInsetFactor = 1
                const insetX = x + (width * (1 - horizontalInsetFactor)) / 2
                const insetY = y
                const insetWidth = width * horizontalInsetFactor
                const insetHeight = height * verticalInsetFactor
                setHeadingRect({ x: insetX, y: insetY, width: insetWidth, height: insetHeight })
            })
        }

        updateRect()

        const resizeObserver = new ResizeObserver(updateRect)
        resizeObserver.observe(panelRef.current)
        resizeObserver.observe(headingRef.current)

        if (typeof window !== "undefined") {
            window.addEventListener("resize", updateRect)
        }

        return () => {
            resizeObserver.disconnect()
            if (typeof window !== "undefined") {
                window.removeEventListener("resize", updateRect)
            }
        }
    }, [])

    const openMenu = useCallback(() => {
        startTransition(() => {
            setIsOpen(true)
        })
        if (onMenuOpen) {
            onMenuOpen()
        }
    }, [onMenuOpen])

    const closeMenu = useCallback(() => {
        startTransition(() => {
            setIsOpen(false)
        })
    }, [])

    const containerStyle: CSSProperties = {
        position: "relative",
        width: "100%",
        height: "100%",
        overflow: "hidden",
        ...style,
    }

    const mainPanelVariants = {
        closed: {
            x: 0,
            y: 0,
            scale: 1,
            rotate: 0,
            borderRadius: 0,
            boxShadow: "0 0 0 rgba(0,0,0,0)",
        },
        open: {
            x: "28%",
            y: "8%",
            scale: 0.65,
            rotate: -14,
            borderRadius: 32,
            boxShadow: "0 40px 80px rgba(0,0,0,0.5)",
        },
    }

    const menuOpacityVariants = {
        closed: { opacity: 0 },
        open: { opacity: 1 },
    }

    const isFixedWidth = style && style.width === "100%"
    const textAlignCenter = { textAlign: "center" as const }

    return (
        <div style={containerStyle}>
            {/* Menu Background */}
            <motion.div
                style={{
                    position: "absolute",
                    inset: 0,
                    backgroundColor: menuBackground,
                    color: primaryTextColor,
                    padding: "32px 56px",
                    boxSizing: "border-box",
                    display: "flex",
                    flexDirection: "column",
                    justifyContent: "flex-start",
                    pointerEvents: isOpen ? "auto" : "none",
                    zIndex: 1,
                    ...(stage2MenuBackgroundImage?.src && {
                        backgroundImage: `url(${stage2MenuBackgroundImage.src})`,
                        backgroundSize: "cover",
                        backgroundPosition: "center",
                    }),
                }}
                variants={menuOpacityVariants}
                animate={isOpen ? "open" : "closed"}
                initial="closed"
                transition={{ duration, ease: [0.22, 0.61, 0.36, 1] }}
            >
                {/* Film grain overlay for menu */}
                <div
                    style={{
                        position: "absolute",
                        inset: 0,
                        backgroundImage: `url("data:image/svg+xml,%3Csvg viewBox='0 0 400 400' xmlns='http://www.w3.org/2000/svg'%3E%3Cfilter id='noiseFilter'%3E%3CfeTurbulence type='fractalNoise' baseFrequency='0.9' numOctaves='4' stitchTiles='stitch'/%3E%3C/filter%3E%3Crect width='100%25' height='100%25' filter='url(%23noiseFilter)' opacity='0.15'/%3E%3C/svg%3E")`,
                        opacity: 0.4,
                        pointerEvents: "none",
                        mixBlendMode: "overlay",
                    }}
                />
                {/* Close Button */}
                <button
                    onClick={closeMenu}
                    style={{
                        background: "none",
                        border: "none",
                        padding: 0,
                        margin: 0,
                        display: "flex",
                        alignItems: "center",
                        gap: 8,
                        cursor: "pointer",
                        color: primaryTextColor,
                        ...buttonFont,
                    }}
                    aria-label={closeLabel}
                >
                    <span
                        aria-hidden="true"
                        style={{
                            fontSize: 20,
                            lineHeight: 1,
                            transform: "translateY(-1px)",
                        }}
                    >
                        ×
                    </span>
                    <span>{closeLabel}</span>
                </button>

                {/* Menu Content */}
                <div
                    style={{
                        flex: 1,
                        display: "flex",
                        flexDirection: "row",
                        alignItems: "center",
                        justifyContent: "space-between",
                        gap: 40,
                    }}
                >
                    {/* Menu Items */}
                    <div
                        style={{
                            maxWidth: "48%",
                            ...headingFont,
                            color: primaryTextColor,
                        }}
                    >
                        <div
                            style={{
                                marginBottom: 40,
                                fontSize: headingFont?.fontSize ?? 32,
                                lineHeight: headingFont?.lineHeight ?? "1.1em",
                                ...textAlignCenter,
                            }}
                        >
                            {introTitle}
                        </div>
                        <div
                            style={{
                                ...bodyFont,
                                display: "flex",
                                flexDirection: "column",
                                gap: 24,
                                fontSize: bodyFont?.fontSize ?? 20,
                            }}
                        >
                            {menuItems.map((item, index) => (
                                <div key={index} style={textAlignCenter}>
                                    {`${index + 1}. ${item}`}
                                </div>
                            ))}
                        </div>
                    </div>

                    {/* Empty spacer so tilted panel on the right has room */}
                    <div style={{ flex: 1, minWidth: 0 }} />
                </div>
            </motion.div>

            {/* Main Panel */}
            <motion.div
                ref={panelRef}
                style={{
                    position: "absolute",
                    inset: 0,
                    backgroundColor: mainBackground,
                    color: inverseTextColor,
                    overflow: "hidden",
                    boxSizing: "border-box",
                    display: "flex",
                    flexDirection: "column",
                    justifyContent: "space-between",
                    transformOrigin: "center center",
                    zIndex: 2,
                }}
                variants={mainPanelVariants}
                initial="closed"
                animate={isOpen ? "open" : "closed"}
                transition={{ duration, ease: [0.22, 0.61, 0.36, 1] }}
            >
                {/* Starfield only on black main panel */}
                {mainBackground === "#000000" && stage === 2 && (
                    <div
                        style={{
                            position: "absolute",
                            inset: 0,
                            zIndex: 0,
                        }}
                    >
                        <TravelInSpace
                            backgroundColor={mainBackground}
                            radiusIn={50}
                            radiusOut={1000}
                            stars={{
                                count: 200,
                                speed: 0.7,
                                thickness: 2,
                                opacity: 0.8,
                                length: 8,
                                thinning: 0.5,
                                shortening: 0.5,
                            }}
                            colors={["#FFFFFF", "#7DD3FC", "#FDE68A"]}
                            style={{ width: "100%", height: "100%" }}
                            clipRect={headingRect || undefined}
                        />
                    </div>
                )}

                {/* Top Bar */}
                <div
                    style={{
                        display: "flex",
                        alignItems: "center",
                        justifyContent: "space-between",
                        padding: "24px 32px",
                        ...buttonFont,
                        position: "relative",
                        zIndex: 1,
                    }}
                >
                    <button
                        onClick={openMenu}
                        style={{
                            display: "flex",
                            alignItems: "center",
                            gap: 8,
                            background: "none",
                            border: "none",
                            color: inverseTextColor,
                            cursor: "pointer",
                            padding: 0,
                            margin: 0,
                            ...buttonFont,
                        }}
                        aria-label={menuLabel}
                    >
                        <span
                            aria-hidden="true"
                            style={{
                                display: "inline-block",
                                position: "relative",
                                width: 18,
                                height: 10,
                            }}
                        >
                            <span
                                style={{
                                    position: "absolute",
                                    top: 0,
                                    left: 0,
                                    right: 0,
                                    height: 2,
                                    backgroundColor: inverseTextColor,
                                }}
                            />
                            <span
                                style={{
                                    position: "absolute",
                                    bottom: 0,
                                    left: 0,
                                    right: 0,
                                    height: 2,
                                    backgroundColor: inverseTextColor,
                                }}
                            />
                        </span>
                        <span>{menuLabel}</span>
                    </button>
                </div>

                {/* Main Content */}
                <div
                    style={{
                        flex: 1,
                        display: "flex",
                        alignItems: "center",
                        justifyContent: "center",
                        padding: "0 40px 40px",
                        boxSizing: "border-box",
                        position: "relative",
                        zIndex: 1,
                    }}
                >
                    {children ? (
                        children
                    ) : (
                        <div
                            ref={headingRef}
                            style={{
                                width: isFixedWidth ? "100%" : "max-content",
                                minWidth: isFixedWidth
                                    ? undefined
                                    : "max-content",
                                ...headingFont,
                                color: inverseTextColor,
                                fontSize: 120,
                                lineHeight: "0.82em",
                                letterSpacing: "-0.04em",
                                textAlign: "center",
                                textTransform: "uppercase",
                            }}
                        >
                            {mainHeadingLines.map((line, index) => (
                                <div key={index}>{line}</div>
                            ))}
                            <div
                                style={{
                                    ...bodyFont,
                                    fontSize: 20,
                                    marginTop: 40,
                                    textTransform: "none",
                                }}
                            >
                                {mainDescription}
                            </div>
                        </div>
                    )}
                </div>
            </motion.div>
        </div>
    )
}

// Starfield component used in CrazyNavigation black panel
function TravelInSpace(props: TravelInSpaceProps) {
    const { backgroundColor, radiusIn, radiusOut, stars, colors, style, clipRect } = props

    const containerRef = useRef<HTMLDivElement | null>(null)
    const canvasRef = useRef<HTMLCanvasElement | null>(null)
    const animationFrameRef = useRef<number | null>(null)
    const starsRef = useRef<Star[]>([])
    const innerRadiusRef = useRef(0)
    const outerRadiusRef = useRef(0)
    const isStatic = useIsStaticRenderer()
    const inView = useInView(containerRef, { margin: "0px", amount: 0.1 })

    const drawColors = useMemo(() => {
        if (!colors || colors.length === 0) return ["#FFFFFF"]
        return colors
    }, [colors])

    const effectiveStars = useMemo<StarsSettings>(() => {
        return {
            count: Math.max(0, Math.floor(stars?.count ?? 150)),
            speed: stars?.speed ?? 0.5,
            thickness: stars?.thickness ?? 2,
            opacity: Math.min(1, Math.max(0, stars?.opacity ?? 0.8)),
            length: stars?.length ?? 8,
            thinning: Math.min(1, Math.max(0, stars?.thinning ?? 0.5)),
            shortening: Math.min(1, Math.max(0, stars?.shortening ?? 0.5)),
        }
    }, [stars])

    useEffect(() => {
        const canvas = canvasRef.current
        const container = containerRef.current
        if (!canvas || !container) return

        const rect = container.getBoundingClientRect()
        const dpr = typeof window !== "undefined" ? window.devicePixelRatio || 1 : 1

        canvas.width = rect.width * dpr
        canvas.height = rect.height * dpr

        const ctx = canvas.getContext("2d")
        if (!ctx) return

        ctx.setTransform(dpr, 0, 0, dpr, 0, 0)

        const maxRadius = Math.hypot(rect.width, rect.height) / 2
        const rawRadiusIn = radiusIn ?? 50
        const rawRadiusOut = radiusOut ?? 1000
        const radiusInNorm = Math.max(0, Math.min(1, rawRadiusIn / Math.max(rawRadiusOut, 1)))
        const innerRadius = maxRadius * radiusInNorm
        const outerRadius = maxRadius

        innerRadiusRef.current = innerRadius
        outerRadiusRef.current = outerRadius

        const { count } = effectiveStars
        const starsArray: Star[] = []

        for (let i = 0; i < count; i++) {
            const angle = Math.random() * Math.PI * 2
            const radius = innerRadius + Math.random() * (outerRadius - innerRadius)
            const speedScale = 0.5 + Math.random() * 0.5
            starsArray.push({ angle, radius, speedScale })
        }

        starsRef.current = starsArray
    }, [effectiveStars, radiusIn, radiusOut, style?.width, style?.height])

    useEffect(() => {
        if (isStatic || !inView) {
            if (animationFrameRef.current !== null && typeof window !== "undefined") {
                cancelAnimationFrame(animationFrameRef.current)
            }
            animationFrameRef.current = null
            return
        }

        const canvas = canvasRef.current
        const container = containerRef.current
        if (!canvas || !container) return

        const ctx = canvas.getContext("2d")
        if (!ctx) return

        let lastTime = performance.now()

        const render = (time: number) => {
            const delta = time - lastTime
            lastTime = time

            const rect = container.getBoundingClientRect()
            const dpr = typeof window !== "undefined" ? window.devicePixelRatio || 1 : 1

            if (canvas.width !== rect.width * dpr || canvas.height !== rect.height * dpr) {
                canvas.width = rect.width * dpr
                canvas.height = rect.height * dpr
                ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
            }

            const width = rect.width
            const height = rect.height
            const cx = width / 2
            const cy = height / 2

            const maxRadius = Math.hypot(width, height) / 2
            const rawRadiusIn = radiusIn ?? 50
            const rawRadiusOut = radiusOut ?? 1000
            const radiusInNorm = Math.max(0, Math.min(1, rawRadiusIn / Math.max(rawRadiusOut, 1)))
            const innerRadius = maxRadius * radiusInNorm
            const outerRadius = maxRadius

            innerRadiusRef.current = innerRadius
            outerRadiusRef.current = outerRadius

            ctx.fillStyle = backgroundColor
            ctx.fillRect(0, 0, width, height)

            if (clipRect) {
                ctx.save()
                ctx.beginPath()
                ctx.rect(0, 0, width, height)
                ctx.rect(clipRect.x, clipRect.y, clipRect.width, clipRect.height)
                ctx.clip("evenodd")
            }

            const { speed, thickness, opacity, length, thinning, shortening } = effectiveStars

            const baseSpeed = speed * 260
            const starLengthBase = length * 3

            const starsArray = starsRef.current
            const count = starsArray.length

            for (let i = 0; i < count; i++) {
                const star = starsArray[i]

                const radiusRange = Math.max(outerRadius - innerRadius, 1)
                const distanceFactor = Math.max(0, Math.min(1, (star.radius - innerRadius) / radiusRange))
                const localSpeed = baseSpeed * star.speedScale * (0.5 + distanceFactor)
                const step = (localSpeed * delta) / 1000

                star.radius += step

                if (star.radius > outerRadius) {
                    star.radius = innerRadius
                    star.angle = Math.random() * Math.PI * 2
                    star.speedScale = 0.5 + Math.random() * 0.5
                }

                const radiusPrev =
                    star.radius -
                    starLengthBase *
                        (1 - shortening * distanceFactor) *
                        (0.5 + 0.5 * distanceFactor)

                const x = cx + Math.cos(star.angle) * star.radius
                const y = cy + Math.sin(star.angle) * star.radius
                const xPrev = cx + Math.cos(star.angle) * radiusPrev
                const yPrev = cy + Math.sin(star.angle) * radiusPrev

                const widthFactor = 1 - thinning * distanceFactor
                const lineWidth = Math.max(0.2, thickness * widthFactor)

                const color = drawColors[i % drawColors.length]

                ctx.globalAlpha = opacity * (0.3 + 0.7 * (1 - distanceFactor))
                ctx.strokeStyle = color
                ctx.lineWidth = lineWidth
                ctx.lineCap = "round"

                ctx.beginPath()
                ctx.moveTo(xPrev, yPrev)
                ctx.lineTo(x, y)
                ctx.stroke()
            }

            if (clipRect) {
                ctx.restore()
            }

            if (typeof window !== "undefined") {
                animationFrameRef.current = requestAnimationFrame(render)
            }
        }

        if (typeof window !== "undefined") {
            animationFrameRef.current = requestAnimationFrame(render)
        }

        return () => {
            if (animationFrameRef.current !== null && typeof window !== "undefined") {
                cancelAnimationFrame(animationFrameRef.current)
            }
            animationFrameRef.current = null
        }
    }, [backgroundColor, effectiveStars, drawColors, radiusIn, radiusOut, isStatic, inView])

    return (
        <div
            ref={containerRef}
            style={{
                position: "relative",
                width: "100%",
                height: "100%",
                overflow: "hidden",
                backgroundColor,
                ...style,
            }}
        >
            <canvas
                ref={canvasRef}
                style={{
                    width: "100%",
                    height: "100%",
                    display: "block",
                }}
                aria-hidden="true"
            />
        </div>
    )
}
